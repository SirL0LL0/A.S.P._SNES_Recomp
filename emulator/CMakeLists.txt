cmake_minimum_required(VERSION 3.20)
project(asp_lab CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Snaggletooth (MIT) — https://github.com/etroimcasso/Snaggletooth, clonato in ./Snaggletooth
set(SNAGGLETOOTH_BUILD_TESTS  OFF CACHE BOOL "" FORCE)
set(SNAGGLETOOTH_BUILD_TOOLS  OFF CACHE BOOL "" FORCE)
set(SNAGGLETOOTH_BUILD_PLAYER OFF CACHE BOOL "" FORCE)
add_subdirectory(Snaggletooth)
add_executable(asp_run asp_run.cpp Snaggletooth/third_party/lodepng/lodepng.cpp)
target_include_directories(asp_run PRIVATE Snaggletooth/third_party/lodepng)
target_link_libraries(asp_run PRIVATE snaggletooth::snaggletooth)
if(MINGW)
  target_link_options(asp_run PRIVATE -static -static-libgcc -static-libstdc++)
endif()
